cmake_minimum_required(VERSION 3.21)
project(voiceforge VERSION 0.3.0 LANGUAGES CXX)
set(voiceforge_package_version "${PROJECT_VERSION}~preview1")

include(CTest)
set(CMAKE_CXX_EXTENSIONS OFF)
set(linux_default OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(linux_default ON)
endif()
option(VOICEFORGE_LV2 "Build the Linux LV2 plugin and bundle" ${linux_default})
option(VOICEFORGE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

if(VOICEFORGE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR MSVC)
        message(FATAL_ERROR "VOICEFORGE_SANITIZERS requires GCC or Clang with GNU-style flags")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_library(voiceforge_dsp STATIC
    src/dsp/core/processor.cpp
    src/dsp/dynamics/voice_dynamics.cpp
    src/dsp/restoration/noise_reduction.cpp)
target_compile_features(voiceforge_dsp PUBLIC cxx_std_20)
target_include_directories(voiceforge_dsp PUBLIC src)
set_target_properties(voiceforge_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)

if(BUILD_TESTING)
    add_executable(voiceforge_dsp_test tests/unit/processor_test.cpp)
    target_link_libraries(voiceforge_dsp_test PRIVATE voiceforge_dsp)
    add_test(NAME dsp COMMAND voiceforge_dsp_test)
    add_executable(voiceforge_voice_test tests/unit/voice_test.cpp)
    target_link_libraries(voiceforge_voice_test PRIVATE voiceforge_dsp)
    add_test(NAME dsp_voice COMMAND voiceforge_voice_test)
    add_executable(voiceforge_restoration_test tests/unit/restoration_test.cpp)
    target_link_libraries(voiceforge_restoration_test PRIVATE voiceforge_dsp)
    add_test(NAME dsp_restoration COMMAND voiceforge_restoration_test)
endif()

if(VOICEFORGE_LV2)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "VOICEFORGE_LV2 requires a Linux target")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LV2 REQUIRED IMPORTED_TARGET lv2>=1.18)
    set(bundle_parent "${CMAKE_CURRENT_BINARY_DIR}/lv2")
    set(bundle "${bundle_parent}/voiceforge.lv2")
    file(MAKE_DIRECTORY "${bundle}")
    foreach(metadata manifest.ttl voiceforge-mono.ttl voiceforge-stereo.ttl)
        configure_file("lv2/${metadata}" "${bundle}/${metadata}" COPYONLY)
    endforeach()
    file(WRITE "${bundle}/VERSION" "${voiceforge_package_version}\n")
    add_library(voiceforge MODULE src/plugin/lv2.cpp)
    target_link_libraries(voiceforge PRIVATE voiceforge_dsp PkgConfig::LV2)
    set_target_properties(voiceforge PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${bundle}"
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)

    if(BUILD_TESTING)
        pkg_check_modules(LILV IMPORTED_TARGET lilv-0)
        if(LILV_FOUND)
            add_executable(voiceforge_host_test tests/integration/host_test.cpp)
            target_compile_features(voiceforge_host_test PRIVATE cxx_std_20)
            target_link_libraries(voiceforge_host_test PRIVATE PkgConfig::LILV PkgConfig::LV2)
            add_dependencies(voiceforge_host_test voiceforge)
            add_test(NAME lv2_host COMMAND voiceforge_host_test "${bundle_parent}")
        endif()
    endif()

    install(TARGETS voiceforge LIBRARY DESTINATION lib/lv2/voiceforge.lv2)
    install(FILES "${bundle}/manifest.ttl" "${bundle}/voiceforge-mono.ttl"
        "${bundle}/voiceforge-stereo.ttl" "${bundle}/VERSION"
        DESTINATION lib/lv2/voiceforge.lv2)
    install(FILES LICENSE DESTINATION share/doc/voiceforge-lv2 RENAME copyright)
    set(CPACK_GENERATOR DEB)
    set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
    set(CPACK_PACKAGE_NAME voiceforge-lv2)
    set(CPACK_PACKAGE_VERSION "${voiceforge_package_version}")
    set(CPACK_PACKAGE_CONTACT "VoiceForge LV2 contributors")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "VoiceForge LV2 offline audio processor preview")
    set(CPACK_DEBIAN_PACKAGE_SECTION sound)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    include(CPack)
endif()
